  @override
  void dispose() {
    _controller.dispose();

  void _analyzeInput(String text) {
    final lowerText = text.toLowerCase();
    final List<EmotionData> detected = [];

    _emotionDatabase.forEach((key, config) {
      int matches = 0;
      for (final keyword in config.keywords) {
        // Word-boundary match to avoid false positives
        // (e.g. "sad" inside "saddle").
        final pattern = RegExp(
          r'\b' + RegExp.escape(keyword) + r'\b',
          caseSensitive: false,
        );
        matches += pattern.allMatches(lowerText).length;
      }
      if (matches > 0) {
        detected.add(EmotionData(config: config, intensity: matches));
      }
    });

    setState(() {
      _activeEmotions = detected;
    });
  }

class EmotionConfig {
  final String label;
  final String emoji;
  final Color color;
  final List<Color> gradient;
  final List<String> keywords;

  EmotionConfig({
    required this.label,
    required this.emoji,
    required this.color,
    required this.gradient,
    required this.keywords,
  });
}

class EmotionData {
  final EmotionConfig config;
  final int intensity;

  EmotionData({required this.config, required this.intensity});
}
